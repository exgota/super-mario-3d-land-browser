        float x, y, z, w;
};

} // namespace math
} // namespace nn

namespace sead
{

template <typename T>
class Matrix34CalcCtr
{
public:
        static void makeQ( nn::math::MTX34& output, const nn::math::QUAT& rotation );
};

} // namespace sead

namespace sead {
template <>
void Matrix34CalcCtr<float>::makeQ( nn::math::MTX34& output, const nn::math::QUAT& rotation )
{
        float xy = rotation.x * rotation.y * 2.0f;
        float xz = rotation.x * rotation.z * 2.0f;
        float xx = rotation.x * rotation.x * 2.0f;
        float yz = rotation.y * rotation.z * 2.0f;
        float yy = rotation.y * rotation.y * 2.0f;
        float wy = rotation.w * rotation.y * 2.0f;
        float wz = rotation.w * rotation.z * 2.0f;
        float zz = rotation.z * rotation.z * 2.0f;
        float wx = rotation.w * rotation.x * 2.0f;

        output.m00 = 1 - yy - zz;
        output.m01 = xy - wz;
        output.m02 = xz + wy;
        output.m03 = 0;
        output.m10 = xy + wz;
        output.m11 = 1 - xx - zz;
        output.m12 = yz - wx;
        output.m13 = 0;
        output.m20 = xz - wy;
        output.m21 = yz + wx;
        output.m22 = 1 - xx - yy;
        output.m23 = 0;
}

}
