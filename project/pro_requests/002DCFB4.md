#include <math/seadVector.h>
#include <math.h>

namespace sead
{

template <typename T>
class MathCalcCommon
{
public:
        static unsigned int atanIdx_( float value );
};

class ControllerBase
{
public:
        unsigned int getStickHold_( unsigned int previous, const Vector2f& stick,
                                    float holdThreshold, float releaseThreshold, int firstBit );
        void updateDerivativeParams_( unsigned int previous, bool pointerOn );

private:
        unsigned int mTriggerMask;
        unsigned int mReleaseMask;
        unsigned int mRepeatMask;
        unsigned int mPointerFlags;
        int mPointerX;
        int mPointerY;
        unsigned char mUnrecovered18[ 0x10 ];
        unsigned int mHoldFrames[ 32 ];
        unsigned char mRepeatDelay[ 32 ];
        unsigned char mRepeatInterval[ 32 ];
        float mLeftStickHoldThreshold;
        float mRightStickHoldThreshold;
        float mLeftStickReleaseThreshold;
        float mRightStickReleaseThreshold;
        int mBitCount;
        int mLeftStickFirstBit;
        int mRightStickFirstBit;
        unsigned int mUnrecovered104[ 2 ];
        unsigned int mHoldMask;
        float mPointerPositionX;
        float mPointerPositionY;
        Vector2f mLeftStick;
        Vector2f mRightStick;
};

static inline void storePointerCoordinate( int* destination, const float* source )
{
        *destination = static_cast<int>( *source );
}

void ControllerBase::updateDerivativeParams_( unsigned int previous, bool pointerOn )
{
        unsigned int stickHold = 0;
        if ( mLeftStickFirstBit >= 0 )
                stickHold = getStickHold_( previous, mLeftStick, mLeftStickHoldThreshold,
                                          mLeftStickReleaseThreshold, mLeftStickFirstBit );
        if ( mRightStickFirstBit >= 0 )
                stickHold |= getStickHold_( previous, mRightStick, mRightStickHoldThreshold,
                                           mRightStickReleaseThreshold, mRightStickFirstBit );

        unsigned int hold = mHoldMask;
        unsigned int stickMask = 0;
        if ( mLeftStickFirstBit >= 0 )
        {
                stickMask = ( 1u << mLeftStickFirstBit ) | ( 1u << ( mLeftStickFirstBit + 1 ) );
                stickMask |= ( 1u << ( mLeftStickFirstBit + 2 ) ) | ( 1u << ( mLeftStickFirstBit + 3 ) );
        }
        if ( mRightStickFirstBit >= 0 )
        {
                stickMask |= 1u << mRightStickFirstBit;
                stickMask |= 1u << ( mRightStickFirstBit + 1 );
                stickMask |= 1u << ( mRightStickFirstBit + 2 );
                stickMask |= 1u << ( mRightStickFirstBit + 3 );
        }
        mHoldMask = ( hold & ~stickMask ) | stickHold;
        mTriggerMask = mHoldMask & ~previous;
        mReleaseMask = previous & ~mHoldMask;
        mRepeatMask = 0;

        for ( int bit = 0; bit < mBitCount; ++bit )
        {
                unsigned int mask = 1u << bit;
                if ( mHoldMask & mask )
                {
                        if ( mRepeatInterval[ bit ] != 0 )
                        {
                                unsigned int delay = mRepeatDelay[ bit ];
                                unsigned int frames = mHoldFrames[ bit ];
                                if ( delay == frames ||
                                     ( delay < frames && ( frames - delay ) % mRepeatInterval[ bit ] == 0 ) )
                                        mRepeatMask |= mask;
                        }
                        ++mHoldFrames[ bit ];
                }
                else
                        mHoldFrames[ bit ] = 0;
        }

        unsigned int pointerFlags;
        if ( !pointerOn && ( mPointerFlags & 1 ) )
                pointerFlags = mPointerFlags | 2;
        else
                pointerFlags = mPointerFlags & ~2u;
        mPointerFlags = pointerFlags;
        if ( pointerOn && !( mPointerFlags & 1 ) )
                pointerFlags = mPointerFlags | 4;
        else
                pointerFlags = mPointerFlags & ~4u;
        mPointerFlags = pointerFlags;
        storePointerCoordinate( &mPointerX, &mPointerPositionX );
        storePointerCoordinate( &mPointerY, &mPointerPositionY );
}

static inline unsigned int calculateStickAngle( float x, float y )
{
        if ( x >= 0.0f )
        {
                if ( y >= 0.0f )
                {
                        if ( x >= y )
                                return MathCalcCommon<float>::atanIdx_( y / x );
                        return 0x40000000u - MathCalcCommon<float>::atanIdx_( x / y );
                }
                float positiveY = -y;
                if ( x >= positiveY )
                        return 0u - MathCalcCommon<float>::atanIdx_( positiveY / x );
                return 0xC0000000u + MathCalcCommon<float>::atanIdx_( x / positiveY );
        }
        if ( y >= 0.0f )
        {
                float positiveX = -x;
                if ( positiveX >= y )
                        return 0x80000000u - MathCalcCommon<float>::atanIdx_( y / positiveX );
                return 0x40000000u + MathCalcCommon<float>::atanIdx_( positiveX / y );
        }
        if ( x <= y )
                return 0x80000000u + MathCalcCommon<float>::atanIdx_( y / x );
        return 0xC0000000u - MathCalcCommon<float>::atanIdx_( x / y );
}

unsigned int ControllerBase::getStickHold_( unsigned int previous, const Vector2f& stick,
                                          float holdThreshold, float releaseThreshold, int firstBit )
{
        float length = sqrtf( stick.x * stick.x + stick.y * stick.y );
        if ( !( length < releaseThreshold ) )
        {
                unsigned int down = 1u << ( firstBit + 1 );
                unsigned int up = 1u << firstBit;
                unsigned int left = 1u << ( firstBit + 2 );
                unsigned int right = 1u << ( firstBit + 3 );
                if ( holdThreshold <= length || ( previous & ( up | down | left | right ) ) )
                {
                if ( stick.x == 0.0f && stick.y == 0.0f )
                        return right;
                unsigned int angle = calculateStickAngle( stick.x, stick.y );
                if ( angle < 0x10000000u )
                        return right;
                if ( angle < 0x30000000u )
                        return right | up;
                if ( angle < 0x50000000u )
                        return up;
                if ( angle < 0x70000000u )
                        return left | up;
                if ( angle < 0x90000000u )
                        return left;
                if ( angle < 0xB0000000u )
                        return left | down;
                if ( angle < 0xD0000000u )
                        return down;
                if ( angle < 0xF0000000u )
                        return right | down;
                return right;
                }
        }
        return 0;
}

} // namespace sead
