# Bounded emission replay

Extract this Python script locally and invoke it with repository path and the two untouched candidate.axf files generated by canonical check.py. Requires capstone and pyelftools. Scope and limitations are in effect-set-emission.md. No binary data is included.

  elif ident==ARM_INS_POP:

  elif ident in (ARM_INS_ADD,ARM_INS_SUB):

  elif ident==ARM_INS_ORR:

  elif ident==ARM_INS_CMP:
   a,b=value(ins,ops[0]),value(ins,ops[1]);x=(a-b)&MASK;n=bool(x&0x80000000);z=x==0;c=a>=b;v=bool(((a^b)&(a^x))&0x80000000)

  elif ident==ARM_INS_B:nxt=value(ins,ops[0])

   elif target==0x292308:
    trace.append(('equal',args[0],args[1]));ret=int(strings[args[0]]==strings[args[1]])
   elif target==0x23f23c:trace.append(('gate',args[0]));ret=gates[args[0]]

   else:raise AssertionError(hex(target))
   r[0]=ret;r[1]=0xcc000001;r[2]=0xcc000002;r[3]=0xcc000003;r[12]=0xcc00000c
  else:raise AssertionError((ins.mnemonic,ins.op_str))
  pc=nxt

rng=random.Random(0x1e98dc);result=[]
for base,end,dirname in [(0x1e98dc,0x1e999c,'function_001E98DC_oc80d277'),(0x1e9a10,0x1e9adc,'function_001E9A10_5l4o1u32')]:

  m={0x1000004:0x2000000,0x1000008:count&MASK,0x1000018:rng.randrange(2),0x100001c:0x3000000,0x1000020:sf,0x1000024:64,0x3000008:0x39e0e4,0x1000028+63:0xff};gates={};expected=[];active=m[0x1000018];emitted=False
  for i in range(max(0,count)):
   ep=0x2100000+i*0x100;cfg=0x2200000+i*0x100;ef=rng.choice([0,*strings]);a,b=rng.randrange(3),rng.randrange(3);gate=rng.randrange(2)

   expected.append(('terminate',));match=ef==sf==0
   if ef and sf:expected.append(('equal',ef,sf));match=strings[ef]==strings[sf]

    expected.append(('gate',ep))

  runs=[run(p,base,m,strings,gates) for p in programs]

print(json.dumps(result,indent=2));Path('/tmp/emission-differential-results.json').write_text(json.dumps(result,indent=2))
